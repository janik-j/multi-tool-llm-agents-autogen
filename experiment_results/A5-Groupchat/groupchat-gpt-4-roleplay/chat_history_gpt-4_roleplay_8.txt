```python
import os
import requests
from datetime import datetime, timedelta
from arxivpy import generate_query, get_articles

# Define the date range
end_date = datetime.now()
start_date = end_date - timedelta(days=3)

# Generate the query
query = generate_query(terms=[''], date_from=start_date, date_until=end_date)

# Get the articles
articles = get_articles(query, start=0, max_results=1000)

# Create the directory if it doesn't exist
if not os.path.exists('/arxiv'):
    os.makedirs('/arxiv')

# Download the PDFs
for article in articles:
    pdf_url = article['pdf_url']
    response = requests.get(pdf_url)
    filename = os.path.join('/arxiv', pdf_url.split('/')[-1])
    with open(filename, 'wb') as f:
        f.write(response.content)
```

Please note that this script will download all articles from arxiv in the last three days, not just PDFs. If you want to download only PDFs, you would need to modify the script to check the file type before downloading. Also, the script assumes that you have permission to write to the /arxiv directory. If you don't, you may need to run the script with sudo or change the directory to a location where you have write permissions.

import yfinance as yf
import pandas as pd
from datetime import datetime, timedelta

# Define the ticker symbol
tickerSymbol = 'AAPL'

# Get data for the last 30 days
endDate = datetime.today().strftime('%Y-%m-%d')
startDate = (datetime.today() - timedelta(days=30)).strftime('%Y-%m-%d')

# Get the data
tickerData = yf.Ticker(tickerSymbol)
tickerDf = tickerData.history(period='1d', start=startDate, end=endDate)

# Calculate the average price
average_price = tickerDf['Close'].mean()

# Save the result to a text file
with open('apple_average.txt', 'w') as f:
    f.write('The average price of AAPL stock in the last 30 days is: $' + str(round(average_price, 2)))

import matplotlib.pyplot as plt
import numpy as np

# Create an array of x values from 0 to 2pi
x = np.linspace(0, 2*np.pi, 100)

# Create an array of y values using numpy's sin function
y = np.sin(x)

# Create a figure and axis
fig, ax = plt.subplots()

# Plot the data
ax.plot(x, y)

# Save the figure

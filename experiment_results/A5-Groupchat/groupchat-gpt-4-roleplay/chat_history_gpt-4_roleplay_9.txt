```python
from github import Github
import operator
import datetime

# First create a Github instance using an access token
g = Github("your_access_token")

# Then play with your Github objects
repo = g.get_repo("microsoft/FLAML")

contributors = {}
for commit in repo.get_commits(since=datetime.datetime.now() - datetime.timedelta(days=30)):
    if commit.committer is not None:
        if commit.committer.login in contributors:
            contributors[commit.committer.login] += 1
        else:
            contributors[commit.committer.login] = 1

# Find the contributor with the most commits
most_active_contributor = max(contributors.items(), key=operator.itemgetter(1))[0]

# Save the result to a text file
with open('flaml_active_contrbutor.txt', 'w') as f:
    f.write(most_active_contributor)
```

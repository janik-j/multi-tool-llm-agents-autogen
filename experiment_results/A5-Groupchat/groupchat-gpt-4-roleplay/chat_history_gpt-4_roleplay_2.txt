```python
import os

# First create a Github instance:

g = Github("access_token")

# Then play with your Github objects:
repo = g.get_repo("dotnet/machinelearning")

open_issues = repo.get_issues(state='open')

with open('mlnet_issue.md', 'w') as f:
    for issue in open_issues[:10]:
        f.write(f"Issue Title: {issue.title}\n")
        f.write(f"Issue Body: {issue.body}\n")
        f.write(f"Issue URL: {issue.html_url}\n")
        f.write("\n------------------------\n")
```

```python
import yfinance as yf
from datetime import datetime, timedelta

# Define the ticker symbol
tickerSymbol = 'AAPL'

# Get data for the last 30 days
end_date = datetime.today().strftime('%Y-%m-%d')
start_date = (datetime.today() - timedelta(days=30)).strftime('%Y-%m-%d')

# Get the data on this ticker
tickerData = yf.Ticker(tickerSymbol)
tickerDf = tickerData.history(period='1d', start=start_date, end=end_date)

# Find the lowest and highest price in the last 30 days
lowest_price = tickerDf['Low'].min()
highest_price = tickerDf['High'].max()

# Calculate the profit
num_stocks = 200
profit = (highest_price - lowest_price) * num_stocks

# Save the result to apple_profit.txt
with open('apple_profit.txt', 'w') as f:
    f.write(f'Profit: ${profit}\n')
```

import requests
from datetime import datetime, timedelta

# Define the repository and the time range
repo = 'Microsoft/FLAML'
since_date = (datetime.now() - timedelta(days=30)).strftime('%Y-%m-%d')

# Make a request to get the contributors
url = f'https://api.github.com/repos/{repo}/stats/contributors'
params = {'since': since_date}
response = requests.get(url, params=params)
contributors = response.json()

# Find the most active contributor
most_active_contributor = max(contributors, key=lambda x: x['total'])

# Save the result to a file
with open('flaml_active_contributor.txt', 'w') as file:
    file.write(f"The most active contributor of FLAML in the last month is: {most_active_contributor['author']['login']}")

print(f"The most active contributor of FLAML in the last month is: {most_active_contributor['author']['login']}")
```
